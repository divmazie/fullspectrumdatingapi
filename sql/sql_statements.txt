-- Deleting duplicate prefs
    SELECT MIN(id) id
    FROM preferences
    GROUP BY dimension_id, profile_id HAVING COUNT(*)>1


    DELETE FROM identities
    WHERE id NOT IN (
        SELECT *
        FROM (
    		SELECT MIN(id) id
        	FROM identities
        	GROUP BY dimension_id, profile_id
        ) temp
    )

    DELETE FROM preferences
    WHERE id NOT IN (
        SELECT *
        FROM (
    		SELECT MIN(id) id
        	FROM preferences
        	GROUP BY dimension_id, profile_id
        ) temp
    )
